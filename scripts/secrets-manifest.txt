# Curio 跨设备依赖清单（哪些东西必须备份）
#
# 格式：每行一个相对路径（相对 content-curator 项目根）或绝对路径。
# 以 # 开头的是注释。
# 备份脚本（backup.sh）会按此清单打包。
# 恢复脚本（setup_new_device.sh）会按此清单还原。

# === 1. 项目内的本地 secret 文件（gitignored，必须单独备份） ===
.gh_pat                       # GitHub PAT，用来 push curio-site 仓库
.smtp_secret                  # QQ 邮箱 SMTP 授权码（自用邮件通知）
worker/.dev.vars              # Cloudflare + Resend + Admin token（CF_TOKEN/RESEND_API_KEY/ADMIN_TOKEN）

# === 2. 用户偏好 + 反馈历史（git 里有但反馈状态会增量更新） ===
profile.yaml                  # 长期身份 + signal_preferences + feedback_timeline

# === 3. 内容数据（候选/打分/导读，git ignore 但有持续价值） ===
topics/                       # 整个目录：candidates/scored/editor_notes/weekly md
.translate_cache/             # 翻译缓存（重建慢，备份省时间）
.article_cache/               # 文章正文缓存（同上）

# === 4. WorkBuddy 调度数据（不在项目里，需要单独导出） ===
# automation 定义保存在 ~/.workbuddy/workbuddy.db 的 automations 表
# 备份脚本会用 sqlite3 导出 INSERT 语句到 workbuddy-automations.sql

# === 5. Python 依赖（venv 不便直接打包，用 pip freeze 生成 requirements） ===
# 备份脚本会跑 pip freeze > scripts/requirements-frozen.txt
